function Block({
  position,
  color = "#6aa84f",
  scale = [1, 1, 1],
}) {
  return (
    <mesh
      position={position}
      scale={scale}
      castShadow
      receiveShadow
    >
      <boxGeometry args={[1, 1, 1]} />
      <meshStandardMaterial color={color} />
    </mesh>
  );
}

// ====================================
// GENERIC BLOCKOUT BUILDING
// ====================================

function Building({
  x,
  z,
  width = 5,
  depth = 5,
  height = 3,
  wallColor = "#777777",
  roofColor = "#555555",
  name = "",
}) {
  const blocks = [];

  // ====================================
  // FLOOR
  // ====================================

  for (let dx = 0; dx < width; dx++) {
    for (let dz = 0; dz < depth; dz++) {
      blocks.push(
        <Block
          key={`floor-${dx}-${dz}`}
          position={[
            x + dx - width / 2,
            0,
            z + dz - depth / 2,
          ]}
          color={wallColor}
        />
      );
    }
  }

  // ====================================
  // WALLS
  // ====================================

  for (let y = 1; y <= height; y++) {

    // Front + Back
    for (let dx = 0; dx < width; dx++) {
      blocks.push(
        <Block
          key={`front-${dx}-${y}`}
          position={[
            x + dx - width / 2,
            y,
            z + depth / 2,
          ]}
          color={wallColor}
        />
      );

      blocks.push(
        <Block
          key={`back-${dx}-${y}`}
          position={[
            x + dx - width / 2,
            y,
            z - depth / 2,
          ]}
          color={wallColor}
        />
      );
    }

    // Left + Right
    for (let dz = 1; dz < depth; dz++) {
      blocks.push(
        <Block
          key={`left-${dz}-${y}`}
          position={[
            x - width / 2,
            y,
            z + dz - depth / 2,
          ]}
          color={wallColor}
        />
      );

      blocks.push(
        <Block
          key={`right-${dz}-${y}`}
          position={[
            x + width / 2,
            y,
            z + dz - depth / 2,
          ]}
          color={wallColor}
        />
      );
    }
  }

  // ====================================
  // ROOF
  // ====================================

  for (
    let dx = -Math.floor(width / 2);
    dx <= Math.floor(width / 2);
    dx++
  ) {
    for (
      let dz = -Math.floor(depth / 2);
      dz <= Math.floor(depth / 2);
      dz++
    ) {
      blocks.push(
        <Block
          key={`roof-${dx}-${dz}`}
          position={[x + dx, height + 1, z + dz]}
          color={roofColor}
        />
      );
    }
  }

  // ====================================
  // BUILDING SIGN
  // ====================================

  if (name) {
    const signZ = z + depth / 2 + 0.8;
    const signY = height + 2;

    blocks.push(
      <group key={`sign-${name}`}>

        {/* Sign board */}
        <Block
          position={[x, signY, signZ]}
          color="#704522"
          scale={[
            Math.max(width * 0.8, 3),
            1.2,
            0.25,
          ]}
        />

        {/* Left post */}
        <Block
          position={[
            x - Math.max(width * 0.8, 3) / 3,
            signY - 1,
            signZ,
          ]}
          color="#4b2e1c"
          scale={[0.18, 1.5, 0.18]}
        />

        {/* Right post */}
        <Block
          position={[
            x + Math.max(width * 0.8, 3) / 3,
            signY - 1,
            signZ,
          ]}
          color="#4b2e1c"
          scale={[0.18, 1.5, 0.18]}
        />

      </group>
    );
  }

  return <>{blocks}</>;
}

// ====================================
// ABOUT HOUSE
// ====================================

function AboutHouse({ x, z }) {
  const blocks = [];

  // Floor
  for (let dx = -3; dx <= 3; dx++) {
    for (let dz = -2; dz <= 2; dz++) {
      blocks.push(
        <Block
          key={`floor-${dx}-${dz}`}
          position={[x + dx, 0, z + dz]}
          color="#704522"
        />
      );
    }
  }

  // Walls
  for (let y = 1; y <= 3; y++) {
    // Front
    for (let dx = -3; dx <= 3; dx++) {
      // Door opening
      if (!(dx === 0 && y <= 2)) {
        blocks.push(
          <Block
            key={`front-${dx}-${y}`}
            position={[x + dx, y, z + 2]}
            color="#9b6a3c"
          />
        );
      }
    }

    // Back
    for (let dx = -3; dx <= 3; dx++) {
      blocks.push(
        <Block
          key={`back-${dx}-${y}`}
          position={[x + dx, y, z - 2]}
          color="#9b6a3c"
        />
      );
    }

    // Left + right
    for (let dz = -1; dz <= 1; dz++) {
      blocks.push(
        <Block
          key={`left-${dz}-${y}`}
          position={[x - 3, y, z + dz]}
          color="#9b6a3c"
        />
      );

      blocks.push(
        <Block
          key={`right-${dz}-${y}`}
          position={[x + 3, y, z + dz]}
          color="#9b6a3c"
        />
      );
    }
  }

  // Door
  blocks.push(
    <Block
      key="door"
      position={[x, 1, z + 2.1]}
      color="#4b2e1c"
      scale={[1, 2, 0.2]}
    />
  );

  // Windows
  blocks.push(
    <Block
      key="window-left"
      position={[x - 1.7, 2, z + 2.1]}
      color="#7ec8e3"
      scale={[1, 1, 0.2]}
    />,

    <Block
      key="window-right"
      position={[x + 1.7, 2, z + 2.1]}
      color="#7ec8e3"
      scale={[1, 1, 0.2]}
    />
  );

  // Roof
  for (let dx = -3; dx <= 3; dx++) {
    for (let dz = -2; dz <= 2; dz++) {
      blocks.push(
        <Block
          key={`roof-${dx}-${dz}`}
          position={[x + dx, 4, z + dz]}
          color="#553522"
        />
      );
    }
  }

  return <>{blocks}</>;
}

// ====================================
// TREE
// ====================================

function Tree({ x, z }) {
  const blocks = [];

  // Trunk
  for (let y = 0; y < 4; y++) {
    blocks.push(
      <Block
        key={`trunk-${x}-${y}-${z}`}
        position={[x, y + 0.5, z]}
        color="#6b3f20"
      />
    );
  }

  const leaves = [
    [-1, 3, -1],
    [0, 3, -1],
    [1, 3, -1],
    [-1, 3, 0],
    [0, 3, 0],
    [1, 3, 0],
    [-1, 3, 1],
    [0, 3, 1],
    [1, 3, 1],
    [0, 4, 0],
  ];

  leaves.forEach(([lx, ly, lz], index) => {
    blocks.push(
      <Block
        key={`leaf-${x}-${z}-${index}`}
        position={[x + lx, ly + 0.5, z + lz]}
        color="#3f7f32"
      />
    );
  });

  return <>{blocks}</>;
}

// ====================================

// ====================================

function Path() {
  const blocks = [];

  // ====================================
  // MAIN NORTH / SOUTH PATH
  // ====================================

  for (let z = -25; z <= 25; z++) {
    blocks.push(
      <Block
        key={`main-${z}`}
        position={[0, 0.05, z]}
        color="#a67c52"
        scale={[2, 0.1, 1]}
      />
    );
  }

  // ====================================
  // PROJECTS PATH
  // ====================================

  for (let x = -1; x >= -17; x--) {
    blocks.push(
      <Block
        key={`projects-${x}`}
        position={[x, 0.05, 0]}
        color="#a67c52"
        scale={[1, 0.1, 2]}
      />
    );
  }

  // ====================================
  // AI LAB PATH
  // ====================================

  for (let x = 1; x <= 17; x++) {
    blocks.push(
      <Block
        key={`lab-${x}`}
        position={[x, 0.05, 0]}
        color="#a67c52"
        scale={[1, 0.1, 2]}
      />
    );
  }

  return <>{blocks}</>;
}

// ====================================
// TERRAIN HEIGHT
// ====================================

function getTerrainHeight(x, z) {
  const hill1 =
    Math.max(
      0,
      4 -
        Math.sqrt(
          (x + 10) ** 2 +
          (z + 10) ** 2
        ) / 3
    );

  const hill2 =
    Math.max(
      0,
      3 -
        Math.sqrt(
          (x - 11) ** 2 +
          (z - 8) ** 2
        ) / 3
    );

  // Keep the central area mostly flat
  const centerDistance =
    Math.sqrt(x * x + z * z);

  if (centerDistance < 7) {
    return 0;
  }

  return Math.floor(
    Math.min(hill1 + hill2, 4)
  );
}

function River() {

  // Main river running across the world
  for (let x = -20; x <= 20; x++) {

    blocks.push(
      <Block
        key={`river-wide-${x}`}
        position={[x, -0.04, 5]}
        color="#4aa3df"
        scale={[1, 0.12, 4]}
      />
    );
  }

  return <>{blocks}</>;
}
